## 2. 契约定稿

### 2.1 契约主键：统一描述符 `ToolDescriptor`

新建 `app/core/tools/descriptor.py`，作为 `ToolEntry` 的超集，是**唯一的运行时工具元数据权威**：

```python
"""统一工具描述符：模块对外暴露的唯一工具契约。"""
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Any, Callable, Optional


@dataclass
class ToolDescriptor:
    """模块对外暴露的工具契约主键。

    - 名称 / schema / coroutine：契约本体
    - source / tool_source / tool_type / category：来源与形态
    - mutating / risk_level / parallel_safe / deferrable / untrusted / verification：
      治理元数据，是 9 处硬编码名单的唯一派生依据
    """
    name: str
    description: str
    args_schema: Any            # pydantic BaseModel 类，可为 None
    coroutine: Callable         # async def execute(**kwargs) -> dict

    # ── 来源 / 形态 ──
    source: str = "builtin"     # 运行时来源（治理唯一权威）
    tool_source: str = "builtin"  # 持久层来源（Tool.tool_source）
    tool_type: str = "code"     # code / http / framework / device
    category: Optional[str] = None

    # ── 治理元数据（派生 9 处名单的唯一依据）──
    mutating: bool = False      # 写操作 → barrier + verification + 结果分类
    risk_level: str = "low"     # low / medium / high / critical（写工具默认 high+）
    parallel_safe: bool = False # 只读并行安全
    deferrable: bool = False    # tool_search 可延迟
    untrusted: bool = False     # 结果不可信包装
    verification: Optional[str] = None  # 写工具验证策略：block_ip/device_action/send_notification/trigger_workflow/None

    # ── 其他 ──
    parameters: Optional[dict] = None
    extra: dict = field(default_factory=dict)  # 扩展位：后续加字段不再大改
```

### 2.3 9 处名单 → 派生函数（统一签名）

新建 `app/core/tools/derive.py`，所有派生函数输入统一为 `registry: dict[str, ToolDescriptor]`，输出 `frozenset[str]`：

```python
"""从 ToolDescriptor 注册表派生治理名单（消除 9 处硬编码）。"""
from __future__ import annotations
from app.core.tools.descriptor import ToolDescriptor

Registry = dict[str, ToolDescriptor]

def write_tool_names(r: Registry) -> frozenset[str]:
    """需要证据链验证的写工具名（替代 WRITE_TOOLS_REQUIRING_VERIFICATION）。"""
    return frozenset(n for n, d in r.items() if d.mutating)

def never_parallel_names(r: Registry) -> frozenset[str]:
    """barrier 工具名（替代 _NEVER_PARALLEL_TOOLS 的「写工具」部分）。"""
    return frozenset(n for n, d in r.items() if d.mutating or not d.parallel_safe)

def parallel_safe_names(r: Registry) -> frozenset[str]:
    """并行安全工具名（替代 _PARALLEL_SAFE_TOOLS 的显式部分）。"""
    return frozenset(n for n, d in r.items() if d.parallel_safe and not d.mutating)

def deferrable_names(r: Registry) -> frozenset[str]:
    """tool_search 可延迟工具名（替代 SOAR_DEFERRABLE_TOOL_SOURCES + 名字名单）。"""
    return frozenset(n for n, d in r.items() if d.deferrable)

def no_effect_names(r: Registry) -> frozenset[str]:
    """无副作用工具名（替代 NO_EFFECT_TOOL_NAMES）。"""
    return frozenset(n for n, d in r.items() if not d.mutating)
```

> 派生函数在 `ToolRegistryBuilder.build()` 完成后调用一次，结果注入 `HermesToolEngine`（作为实例属性），`tool_dispatch`/`verification`/`tool_result_classification`/`tool_search` 改为读取这些派生集合，不再 import 各自的常量。**过渡期**：先保留各文件的常量作为 fallback，逐步替换。

**③ 新增两个 helper（追加到文件末尾，`_find_tool_result` 之后）：**

```python
def _entry_ip(entry: dict) -> Optional[str]:
    """从 tool_history 条目提取「被查 IP」。
    优先 arguments.ip（LLM 实际传入的查询参数），回退 result.ip（工具回显）。
    """
    args = entry.get("arguments") or {}
    ip = args.get("ip") or args.get("target_ip")
    if ip:
        return str(ip)
    result = entry.get("result")
    if isinstance(result, dict):
        ip = result.get("ip")
        if ip:
            return str(ip)
    return None


def _find_tool_entry_for_ip(tool_history: list[dict], name: str, ip: str) -> Optional[dict]:
    """反查「被查 IP == 目标 IP」的最近一条某工具记录；不存在返回 None。"""
    for entry in reversed(tool_history):
        if entry.get("name") != name:
            continue
        if _entry_ip(entry) == ip:
            return entry
    return None
```

**④ 重写 `_verify_block_ip`（:133-195）整段：**

```python
def _verify_block_ip(
    arguments: dict,
    context: dict,
    config: VerificationConfig,
) -> VerificationResult:
    """验证 block_ip_on_firewall 的证据链（证据必须绑定目标 IP）。"""
    missing: list[str] = []
    ip = arguments.get("ip") or arguments.get("target_ip")
    if not ip:
        missing.append("目标 IP 未指定")
    elif not _is_valid_ip(ip):
        missing.append(f"目标 IP 格式不合法: {ip}")
    else:
        ip = str(ip)

    tool_history: list[dict] = context.get("tool_history", [])
    valid_ip = isinstance(ip, str) and ip

    # whitelist 检查（证据绑定目标 IP）
    wl_entry = _find_tool_entry_for_ip(tool_history, "check_whitelist", ip) if valid_ip else None
    if wl_entry is None:
        if any(e.get("name") == "check_whitelist" for e in tool_history):
            missing.append(f"白名单查询的 IP 与封禁目标 IP（{ip}）不一致，证据未绑定")
        else:
            missing.append("未查询白名单（应先调用 check_whitelist 确认 IP 不在白名单）")
    else:
        wl_result = wl_entry.get("result")
        if isinstance(wl_result, dict) and wl_result.get("in_whitelist"):
            return VerificationResult(
                approved=False,
                code="blocked",
                message=f"IP {ip} 在白名单中，禁止封禁。",
                missing_evidence=missing,
            )

    # asset 确认（证据绑定目标 IP）
    asset_entry = _find_tool_entry_for_ip(tool_history, "get_asset_info", ip) if valid_ip else None
    if asset_entry is None:
        if any(e.get("name") == "get_asset_info" for e in tool_history):
            missing.append(f"资产查询的 IP 与封禁目标 IP（{ip}）不一致，证据未绑定")
        else:
            missing.append("未查询资产信息（应先调用 get_asset_info 确认 IP 属于已知资产）")

    # threat_intel（可选，同样绑定 IP）
    if config.require_threat_intel:
        ti_entry = _find_tool_entry_for_ip(tool_history, "get_threat_intel", ip) if valid_ip else None
        if ti_entry is None:
            missing.append("未查询威胁情报（require_threat_intel=True 时必须先查 get_threat_intel）")

    # duration 校验
    duration = arguments.get("duration")
    if duration is not None:
        if not _is_valid_duration(duration):
            missing.append(f"封禁时长不合法: {duration}")

    if missing:
        return VerificationResult(
            approved=False,
            code="needs_clarification" if len(missing) <= 2 else "blocked",
            message="封禁前证据不充分: " + "; ".join(missing),
            missing_evidence=missing,
            clarification_question=f"是否确认要封禁 IP {ip}？证据不足: {'; '.join(missing)}",
            clarification_choices=["确认封禁", "取消，先补充证据"],
        )

    return VerificationResult(approved=True, code="approved", message="block_ip 证据充分")
```

from __future__ import annotations
from typing import Any, Optional
from app.core.tools.descriptor import ToolDescriptor


class ToolRegistryBuilder:
    def __init__(self, db, agent, *, user: Optional[Any] = None):
        self.db = db
        self.agent = agent
        self.user_id = getattr(user, "id", None) if user is not None else None
        self._preset_index = None  # 懒加载 ToolCatalog.preset_name_index()

    def build(self) -> dict[str, ToolDescriptor]:
        registry: dict[str, ToolDescriptor] = {}
        self._register_db_tools(registry)
        self._register_kb_tools(registry)
        self._register_asset_tools(registry)
        self._register_workflow_tools(registry)
        # 阶段 2：self._register_device_tools(registry)
        return registry

    # ── Provider（每个只产出 ToolDescriptor，不碰治理名单）──
    def _register_db_tools(self, registry): ...
    def _register_kb_tools(self, registry): ...
    def _register_asset_tools(self, registry): ...
    def _register_workflow_tools(self, registry): ...

    # ── 治理元数据解析 ──
    def _preset_meta(self, name: str) -> dict:
        """预设工具从种子 dict 读 mutating/risk_level/parallel_safe/deferrable。"""
        if self._preset_index is None:
            from app.core.tools.catalog import ToolCatalog
            self._preset_index = ToolCatalog.preset_name_index()

def _entry(name, ip, result):
